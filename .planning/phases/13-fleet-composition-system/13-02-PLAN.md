1. Cache DOM references on first call:

```javascript
let _fleetCalloutEls = null;
function _ensureFleetCalloutEls() {
  if (_fleetCalloutEls) return;
  _fleetCalloutEls = [
    document.getElementById('fleet-callout-0'),
    document.getElementById('fleet-callout-1'),
    document.getElementById('fleet-callout-2')
  ];
  // Setup animationend listeners (same pattern as wave announce)
  for (const el of _fleetCalloutEls) {
    if (el) el.addEventListener('animationend', function() { this.classList.remove('visible'); });
  }
}
```

2. `showFleetCallouts(fleetResults)` function:
```javascript
function showFleetCallouts(fleetResults) {
  _ensureFleetCalloutEls();
  if (!_fleetCalloutEls) return;
  for (let i = 0; i < 3; i++) {
    const el = _fleetCalloutEls[i];
    if (!el) continue;
    if (i >= fleetResults.length) { el.classList.remove('visible'); continue; }
    const fr = fleetResults[i];
    // Set callout text
    el.querySelector('.fleet-callout-text').textContent = fr.callout;
    // Boss class toggle
    if (fr.isBoss) el.classList.add('boss'); else el.classList.remove('boss');
    // Compute directional chevron angle: player-to-fleet centroid
    const dx = fr.centroidX - flyPos[0];
    const dz = fr.centroidZ - flyPos[2];
    const angle = Math.atan2(dx, dz); // world-space angle
    const chevron = el.querySelector('.fleet-callout-chevron');
    chevron.style.transform = 'rotate(' + ((angle * 180 / Math.PI) + 180) + 'deg)';
    // Stagger animation: delay = i * 1.2s
    el.style.animationDelay = (i * 1.2) + 's';
    // Trigger animation (reflow trick)
    el.classList.remove('visible');
    void el.offsetWidth;
    el.classList.add('visible');
  }
}
```
