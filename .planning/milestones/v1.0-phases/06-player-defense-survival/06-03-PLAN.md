2. Create `updateEnemyIndicators()` function, called each frame in the render loop (HUD update section), gated on `flyMode && playerState.alive && playerState.deathPhase === 0`:

```javascript
function updateEnemyIndicators() {
  const margin = 40; // pixels from viewport edge
  const cx = baseWidth / 2, cy = baseHeight / 2;

  for (let i = 0; i < MAX_ENEMIES; i++) {
    if (!enemies.alive[i]) {
      indicatorEls[i].style.display = 'none';
      continue;
    }

    // Distance check
    const dx = enemies.posX[i] - flyPos[0];
    const dz = enemies.posZ[i] - flyPos[2];
    const dist = Math.sqrt(dx * dx + dz * dz);
    if (dist > INDICATOR_RANGE) {
      indicatorEls[i].style.display = 'none';
      continue;
    }

    // Project enemy world position to screen
    const wx = enemies.posX[i], wy = enemies.posY[i], wz = enemies.posZ[i];
    const clipX = _mvp[0]*wx + _mvp[4]*wy + _mvp[8]*wz + _mvp[12];
    const clipY = _mvp[1]*wx + _mvp[5]*wy + _mvp[9]*wz + _mvp[13];
    const clipW = _mvp[3]*wx + _mvp[7]*wy + _mvp[11]*wz + _mvp[15];

    if (clipW <= 0.01) {
      // Behind camera -- place indicator on opposite side of screen
      // Use the angle from screen center toward where the enemy would be "behind"
      const angle = Math.atan2(-clipY, -clipX);
      const edgeX = cx + Math.cos(angle) * (cx - margin);
      const edgeY = cy - Math.sin(angle) * (cy - margin); // Y inverted
      positionIndicator(indicatorEls[i], edgeX, edgeY, angle, dist, enemies.type[i]);
      continue;
    }

    const ndcX = clipX / clipW;
    const ndcY = clipY / clipW;
    const sx = (ndcX * 0.5 + 0.5) * baseWidth;
    const sy = (1.0 - (ndcY * 0.5 + 0.5)) * baseHeight;

    // Check if on-screen (with margin)
    const onScreen = sx >= margin && sx <= baseWidth - margin &&
                     sy >= margin && sy <= baseHeight - margin;

    if (onScreen) {
      indicatorEls[i].style.display = 'none';
      continue;
    }

    // Off-screen: clamp to edge
    const angle = Math.atan2(sy - cy, sx - cx);
    // Compute edge intersection
    const abscos = Math.abs(Math.cos(angle));
    const abssin = Math.abs(Math.sin(angle));
    let edgeX, edgeY;
    if (abscos * (baseHeight - 2 * margin) > abssin * (baseWidth - 2 * margin)) {
      // Intersects left/right edge
      edgeX = cx + Math.sign(Math.cos(angle)) * (cx - margin);
      edgeY = cy + Math.tan(angle) * Math.sign(Math.cos(angle)) * (cx - margin);
      edgeY = Math.max(margin, Math.min(baseHeight - margin, edgeY));
    } else {
      // Intersects top/bottom edge
      edgeY = cy + Math.sign(Math.sin(angle)) * (cy - margin);
      edgeX = cx + (1 / Math.tan(angle)) * Math.sign(Math.sin(angle)) * (cy - margin);
      edgeX = Math.max(margin, Math.min(baseWidth - margin, edgeX));
    }

    positionIndicator(indicatorEls[i], edgeX, edgeY, angle, dist, enemies.type[i]);
  }
}

function positionIndicator(el, x, y, angle, dist, enemyType) {
  el.style.display = 'block';
  el.style.left = x + 'px';
  el.style.top = y + 'px';
  // Rotate chevron to point toward enemy (angle is from center to edge)
  const rotDeg = (angle * 180 / Math.PI) + 90; // +90 because chevron points up by default
  el.querySelector('.indicator-chevron').style.transform = 'rotate(' + rotDeg + 'deg)';
  el.querySelector('.indicator-dist').textContent = Math.floor(dist) + 'u';
  // Color by archetype (default red for Grunt type=0)
  const colors = ['rgba(255,60,40,0.9)', 'rgba(60,255,140,0.9)', 'rgba(255,200,40,0.9)', 'rgba(140,60,255,0.9)', 'rgba(255,140,40,0.9)'];
  const color = colors[enemyType] || colors[0];
  el.style.color = color;
}
```
